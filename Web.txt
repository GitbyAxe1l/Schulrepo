  <script>
    // Grobe Standardwerte für DAS Rezept
    // kcal/100g (oder pro Stück bei Eier) -> du kannst die Zahlen easy anpassen.
    const ingredients = [
      { key: "butter", label: "Butter", unit: "g", amount: 120, kcalPer100g: 717 },
      { key: "choc", label: "Zartbitterschokolade", unit: "g", amount: 120, kcalPer100g: 550 },
      { key: "sugar", label: "Zucker", unit: "g", amount: 120, kcalPer100g: 400 },
      { key: "eggs", label: "Eier", unit: "Stk", amount: 2, kcalEach: 78 },
      { key: "flour", label: "Mehl", unit: "g", amount: 100, kcalPer100g: 364 },
      { key: "cocoa", label: "Kakao", unit: "g", amount: 25, kcalPer100g: 228 },
      { key: "bp", label: "Backpulver", unit: "TL", amount: 1, kcalEach: 5 },
      { key: "salt", label: "Salz", unit: "Prise", amount: 1, kcalEach: 0 }
    ];

    const kcalBody = document.getElementById("kcalBody");
    const piecesEl = document.getElementById("pieces");
    const totalKcalEl = document.getElementById("totalKcal");
    const perPieceEl = document.getElementById("perPieceKcal");
    const recalcBtn = document.getElementById("recalc");
    const resetBtn = document.getElementById("reset");

    function kcalFor(ing){
      if (ing.kcalEach != null) return ing.amount * ing.kcalEach;
      return (ing.amount / 100) * ing.kcalPer100g;
    }

    function format(n){
      return Math.round(n);
    }

    function renderTable(){
      kcalBody.innerHTML = "";
      ingredients.forEach(ing => {
        const tr = document.createElement("tr");
        tr.className = "kcal-row";

        const td1 = document.createElement("td");
        td1.innerHTML = `<b>${ing.label}</b><div class="tag">${ing.unit}</div>`;

        const td2 = document.createElement("td");
        td2.style.textAlign = "right";
        td2.innerHTML = `
          <input
            type="number"
            min="0"
            step="${ing.unit === 'Stk' || ing.unit === 'TL' || ing.unit === 'Prise' ? '1' : '5'}"
            value="${ing.amount}"
            data-key="${ing.key}"
            aria-label="${ing.label} Menge"
          />
        `;

        const td3 = document.createElement("td");
        td3.style.textAlign = "right";
        td3.innerHTML = `<b><span id="k_${ing.key}">${format(kcalFor(ing))}</span> kcal</b>`;

        tr.appendChild(td1);
        tr.appendChild(td2);
        tr.appendChild(td3);
        kcalBody.appendChild(tr);
      });

      // Inputs live verdrahten
      kcalBody.querySelectorAll("input[type='number']").forEach(inp => {
        inp.addEventListener("input", () => {
          const key = inp.dataset.key;
          const ing = ingredients.find(x => x.key === key);
          ing.amount = Number(inp.value || 0);
          update();
        });
      });
    }

    function update(){
      // pro Zutat aktualisieren
      let total = 0;
      ingredients.forEach(ing => {
        const k = kcalFor(ing);
        total += k;
        const el = document.getElementById("k_" + ing.key);
        if (el) el.textContent = format(k);
      });

      const pieces = Math.max(1, Number(piecesEl.value || 1));
      const perPiece = total / pieces;

      totalKcalEl.textContent = format(total);
      perPieceEl.textContent = format(perPiece);
    }

    recalcBtn.addEventListener("click", update);
    resetBtn.addEventListener("click", () => {
      // zurück auf Standardmengen
      const defaults = {
        butter:120, choc:120, sugar:120, eggs:2, flour:100, cocoa:25, bp:1, salt:1
      };
      ingredients.forEach(ing => ing.amount = defaults[ing.key]);
      piecesEl.value = 10;

      // inputs in der Tabelle wieder setzen
      kcalBody.querySelectorAll("input[type='number']").forEach(inp => {
        const key = inp.dataset.key;
        inp.value = defaults[key];
      });

      update();
    });

    piecesEl.addEventListener("input", update);

    renderTable();
    update();
  </script>
