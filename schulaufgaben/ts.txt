using System;

class Program
{
    static void Main()
    {
        Console.Write("Rechnung eingeben: ");
        string eingabe = Console.ReadLine();

        eingabe = eingabe.Replace(" ", "");
        eingabe += "+"; // damit letzte Zahl verarbeitet wird

        int ergebnis = 0;
        string zahlText = "";
        char letzterOperator = '+';

        bool neueZahl = true; // wichtig!

        for (int i = 0; i < eingabe.Length; i++)
        {
            char zeichen = eingabe[i];

            // ZAHL oder VORZEICHEN
            if (char.IsDigit(zeichen) || (zeichen == '-' && neueZahl))
            {
                zahlText += zeichen;
                neueZahl = false;
            }
            // OPERATOR
            else if (zeichen == '+' || zeichen == '-' || zeichen == '*' || zeichen == '/')
            {
                int zahl = int.Parse(zahlText);

                if (letzterOperator == '+')
                    ergebnis = ergebnis + zahl;
                else if (letzterOperator == '-')
                    ergebnis = ergebnis - zahl;
                else if (letzterOperator == '*')
                    ergebnis = ergebnis * zahl;
                else if (letzterOperator == '/')
                    ergebnis = ergebnis / zahl;

                letzterOperator = zeichen;
                zahlText = "";
                neueZahl = true; // neue Zahl beginnt!
            }
        }

        Console.WriteLine("Ergebnis: " + ergebnis);
    }
}
